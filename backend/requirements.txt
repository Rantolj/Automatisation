# -----------------------------------------------------------------------------
# FRAMEWORK WEB
# -----------------------------------------------------------------------------
# FastAPI : Framework web moderne, rapide, avec validation automatique
# Justification : 
#   - Plus performant que Flask (basé sur Starlette)
#   - Documentation auto-générée (Swagger/OpenAPI)
#   - Validation native avec Pydantic
#   - Support async natif
#   - Typage Python moderne
fastapi==0.109.0

# -----------------------------------------------------------------------------
# SERVEUR ASGI
# -----------------------------------------------------------------------------
# Uvicorn : Serveur ASGI performant pour exécuter FastAPI
# Justification :
#   - Recommandé officiellement pour FastAPI
#   - Support async/await
#   - Rechargement à chaud en développement
uvicorn[standard]==0.27.0

# -----------------------------------------------------------------------------
# VALIDATION (inclus avec FastAPI, mais explicité)
# -----------------------------------------------------------------------------
# Pydantic : Validation et sérialisation de données
# Justification :
#   - Requis par FastAPI
#   - Validation de schémas JSON
#   - Conversion automatique de types
pydantic==2.5.3

# -----------------------------------------------------------------------------
# UPLOAD DE FICHIERS
# -----------------------------------------------------------------------------
# python-multipart : Parsing des requêtes multipart/form-data
# Justification :
#   - Requis pour les uploads de fichiers avec FastAPI
#   - Gère les formulaires avec fichiers
python-multipart==0.0.6
